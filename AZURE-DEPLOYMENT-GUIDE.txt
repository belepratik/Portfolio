AZURE DEPLOYMENT GUIDE - FREE TIER
===================================

🎯 AZURE STRATEGY: Container Instances + Azure Database + Static Web Apps
- Backend: Azure Container Instances (Free tier: $0 for first month)
- Database: Azure Database for MySQL (Free tier: B1ms, 32GB storage)
- Frontend: Azure Static Web Apps (Free tier: 100GB bandwidth)

📋 PREREQUISITES
----------------
1. Azure Account (free tier eligible)
2. Azure CLI installed
3. Docker installed locally

⚡ STEP-BY-STEP AZURE DEPLOYMENT
===============================

🔧 1. SETUP AZURE CLI
---------------------
# Install Azure CLI
https://docs.microsoft.com/en-us/cli/azure/install-azure-cli

# Login to Azure
az login

# Set subscription (if you have multiple)
az account set --subscription "your-subscription-id"

🗄️ 2. CREATE AZURE DATABASE FOR MYSQL
--------------------------------------
# Create resource group
az group create --name portfolio-rg --location eastus

# Create MySQL server (Free tier)
az mysql flexible-server create \
  --resource-group portfolio-rg \
  --name portfolio-mysql-server \
  --admin-user portfolioadmin \
  --admin-password "YourSecurePassword123!" \
  --sku-name Standard_B1ms \
  --tier Burstable \
  --storage-size 32 \
  --version 8.0.21 \
  --location eastus

# Create database
az mysql flexible-server db create \
  --resource-group portfolio-rg \
  --server-name portfolio-mysql-server \
  --database-name portfolio_db

# Configure firewall (allow Azure services)
az mysql flexible-server firewall-rule create \
  --resource-group portfolio-rg \
  --name portfolio-mysql-server \
  --rule-name AllowAzureServices \
  --start-ip-address 0.0.0.0 \
  --end-ip-address 0.0.0.0

🐳 3. DEPLOY BACKEND TO AZURE CONTAINER REGISTRY
-----------------------------------------------
# Create container registry
az acr create \
  --resource-group portfolio-rg \
  --name portfolioregistry \
  --sku Basic

# Build and push to ACR
az acr build \
  --registry portfolioregistry \
  --image portfolio-backend:latest \
  ./backend

🚀 4. DEPLOY TO AZURE CONTAINER INSTANCES
-----------------------------------------
# Get MySQL connection string
MYSQL_HOST=$(az mysql flexible-server show --resource-group portfolio-rg --name portfolio-mysql-server --query "fullyQualifiedDomainName" -o tsv)

# Deploy container instance
az container create \
  --resource-group portfolio-rg \
  --name portfolio-backend-container \
  --image portfolioregistry.azurecr.io/portfolio-backend:latest \
  --cpu 1 \
  --memory 1 \
  --registry-login-server portfolioregistry.azurecr.io \
  --registry-username portfolioregistry \
  --registry-password $(az acr credential show --name portfolioregistry --query "passwords[0].value" -o tsv) \
  --ip-address Public \
  --ports 8080 \
  --environment-variables \
    DB_HOST="$MYSQL_HOST" \
    DB_USERNAME="portfolioadmin" \
    DB_PASSWORD="YourSecurePassword123!" \
    DB_NAME="portfolio_db" \
    SERVER_PORT="8080"

# Get backend URL
az container show --resource-group portfolio-rg --name portfolio-backend-container --query ipAddress.fqdn

🌐 5. DEPLOY FRONTEND TO AZURE STATIC WEB APPS
----------------------------------------------
# Install Static Web Apps CLI
npm install -g @azure/static-web-apps-cli

# Build frontend with backend URL
cd frontend
echo "VITE_API_URL=http://[your-backend-fqdn]:8080/api" > .env.production
npm run build

# Deploy to Static Web Apps via GitHub
# Go to Azure Portal > Static Web Apps > Create
# Connect GitHub repository
# Build settings:
#   - App location: /frontend
#   - Build location: dist
#   - Output location: dist

# Alternative: Deploy directly
az staticwebapp create \
  --name portfolio-frontend \
  --resource-group portfolio-rg \
  --source https://github.com/yourusername/Portfolio \
  --location "East US 2" \
  --branch featue/NewVersion \
  --app-location "/frontend" \
  --output-location "dist"

📱 6. GET YOUR AZURE URLS
------------------------
Backend: http://[container-fqdn]:8080
Frontend: https://[staticwebapp-name].azurestaticapps.net
